\section{Resumo}


\begin{frame}{Resumo}
\begin{itemize}
\item Modelo: \textbf{TSDF\_seqKANSeq} em modo \textbf{KAN-only} com \textbf{rebuild} (reconstru\c{c}\~ao direta).
\item Entrada observada por tempo:
\[
x_t \in \mathbb{R}^{C}, \qquad \text{mask}_t \in \{0,1\}^{C}
\]
\item A m\'ascara modula a mem\'oria:
\[
z_t = \mathrm{KAN}_{enc}(x_t), \qquad \gamma_t = W(1-\text{mask}_t),
\qquad m_t = \gamma_t \odot h_{t-1}
\]
\item Atualiza\c{c}\~ao recorrente:
\[
h_t = \mathrm{KAN}_{cell}\big([\,z_t,\; h_{t-1},\; m_t\,]\big)
\]
\item A sa\'ida \'e a reconstru\c{c}\~ao do pr\'oprio $x_t$:
\[
\hat{x}_t = \mathrm{KAN}_{out}(h_t)
\]
\item Treino por reconstru\c{c}\~ao (MSE).
\end{itemize}
\end{frame}

\begin{frame}{Resumo}
\begin{itemize}
\item O modelo combina \textbf{entrada atual} com \textbf{mem\'oria passada} para reconstruir a s\'erie.
\item A entrada $x_t$ passa por um KAN (\textbf{KAN\_enc}) para gerar um latente $z_t$.
\item A m\'ascara indica faltantes e \textbf{modula} a mem\'oria:
$m_t$ refor\c{c}a/atenua partes de $h_{t-1}$.
\item A c\'elula recorrente \textbf{junta} $z_t$, $h_{t-1}$ e $m_t$ para produzir o novo estado $h_t$.
\item A sa\'ida $\hat{x}_t$ \'e lida do estado via \textbf{KAN\_out}.
\end{itemize}
\end{frame}

\section{Metodologia}

\begin{frame}{ArticleKAN: splines por conex\~ao}
\small
\begin{itemize}
\item O backbone usa o \textbf{ArticleKAN}: cada aresta $i\!\to\! j$ aprende uma spline univariada.
\item Forma geral:
\[
y_j = \sum_i f_{ij}(x_i), \qquad f_{ij} \text{ spline 1D}
\]
\item Cada spline \'{e} definida por:
\begin{itemize}
\item \textbf{grid} (n\'os) e ordem $k$
\item coeficientes das bases B-spline
\end{itemize}
\item Vantagens:
\begin{itemize}
\item interpretabilidade por conex\~ao (curva $f_{ij}$)
\item controle estrutural da malha de conex\~oes
\item n\~ao-linearidade local por partes
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Vis\~ao geral: SeqKAN recorrente}
\small
Arquitetura por passo de tempo $t$ (com $H=\texttt{hidden\_size}$)
\[
\boxed{
\begin{aligned}
z_t &= \mathrm{KAN}_{enc}(x_t) \in \mathbb{R}^{H} \\
\gamma_t &= W(1-\mathrm{mask}_t) + b \in \mathbb{R}^{H} \\
m_t &= \gamma_t \odot h_{t-1} \in \mathbb{R}^{H} \\
h_t &= \mathrm{KAN}_{cell}\big([z_t,\; h_{t-1},\; m_t]\big) \in \mathbb{R}^{H} \\
\hat{x}_t &= \mathrm{KAN}_{out}(h_t) \in \mathbb{R}^{C}
\end{aligned}}
\]

\begin{itemize}
\item \textbf{KAN\_enc}: projeta $x_t$ (dim $C$) para o espa\c{c}o latente $H$ usando splines por conex\~ao.
\item \textbf{KAN\_cell}: recebe um vetor de dimens\~ao $3H$ e atualiza a mem\'oria recorrente.
\item \textbf{KAN\_out}: head de reconstru\c{c}\~ao $\hat{x}_t$ a partir do estado.
\end{itemize}
\end{frame}




% --------------------------------------------------
\begin{frame}{M\'ascara $\rightarrow$ mem\'oria modulada}
\small

A m\'ascara gera um gate em hidden via proje\c{c}\~ao linear:
\[
\gamma_t = W(1-\mathrm{mask}_t) + b
\]
e modula o estado anterior:
\[
m_t = \gamma_t \odot h_{t-1}.
\]
Este termo $m_t$ entra como \textbf{terceiro bloco} da entrada da c\'elula:
\[
h_t = \mathrm{KAN}_{cell}([z_t,\; h_{t-1},\; m_t]).
\]

\end{frame}

% =========================
% (4) Top-k — corrigido para o que existe (activation top-k)
% =========================
\begin{frame}{Top-k: esparsidade por ativa\c{c}\~ao ($k_h$)}
\small
\begin{itemize}
\item Existe top-k \textbf{por ativa\c{c}\~ao} aplicado aos vetores $z_t$ e $h_t$:
\[
z_t \leftarrow \mathrm{TopK}(z_t; k_h), \qquad h_t \leftarrow \mathrm{TopK}(h_t; k_h)
\]
mantendo apenas as $k_h$ maiores componentes em magnitude ($|\cdot|$) por amostra/tempo.
\item Warmup:O top-k s\'o entra ap\'os \texttt{warmup\_epochs}.

\item \textbf{Bottleneck din\^amico} no hidden, reduzindo complexidade e aumentando interpretabilidade por dimens\~ao ativa.
\end{itemize}



\end{frame}

% --------------------------------------------------
\begin{frame}{Interpretabilidade no KAN: leitura por aresta (edge-wise)}
\small
\begin{itemize}
\item Em KAN, cada conex\~ao $i\!\to\! j$ \'{e} uma fun\c{c}\~ao univariada aprendida $f_{ij}(\cdot)$ (spline):
\[
y_j = \sum_i f_{ij}(u_i)
\]
\item No modelo atual existem tr\^es conjuntos de fun\c{c}\~oes inspecion\'aveis:
\begin{itemize}
\item $\mathrm{KAN}_{enc}$: $f^{enc}_{c\to j}(x_{t,c})$ (feature $\to$ hidden)
\item $\mathrm{KAN}_{cell}$: $f^{cell}_{j\to r}([z,h,m]_{t,j})$ (din\^amica no hidden)
\item $\mathrm{KAN}_{out}$: $f^{out}_{j\to c}(h_{t,j})$ (hidden $\to$ feature reconstru\'ida)
\end{itemize}
\item Estrat\'egia de explica\c{c}\~ao recomendada: selecionar arestas mais relevantes (por $|\text{coef}|$ ou contribui\c{c}\~ao) e plotar suas splines.
\end{itemize}
\end{frame}

% (6) Conclusão
% =========================
\begin{frame}{Conclus\~ao (alinhada ao c\'odigo atualizado)}
O TSDF\_seqKANSeq (KAN-only) implementa reconstru\c{c}\~ao recorrente com tr\^es KANs:
\[
z_t=\mathrm{KAN}_{enc}(x_t),\quad
\]
\[
m_t=\gamma_t\odot h_{t-1},\quad
\]
\[
h_t=\mathrm{KAN}_{cell}([z_t,h_{t-1},m_t]),\quad
\]
\[
\hat{x}_t=\mathrm{KAN}_{out}(h_t)
\]

\begin{itemize}
\item N\~ao-linearidade distribu\'ida nas arestas (B-splines do \texttt{ArticleKAN}) em \textbf{encoder, c\'elula e sa\'ida}.
\item M\'ascara atua como \textbf{condicionamento} via $m_t$ (mem\'oria modulada).
\item Esparsidade: \textbf{Top-k por ativa\c{c}\~ao} em $z_t$ e $h_t$ (par\^ametro $k_h$); top-k estrutural est\'a desligado.
\item Difus\~ao de missingness aplicada.
\end{itemize}
\end{frame}
