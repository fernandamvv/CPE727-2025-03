\section{Resumo}


\begin{frame}{Resumo executivo}
\begin{itemize}
\item Modelo: \textbf{TSDF\_seqKANSeq} com \textbf{KAN\_cell + head linear} e \textbf{rebuild} (reconstru\c{c}\~ao direta).
\item Entrada observada por tempo:
\[
x_t \in \mathbb{R}^{C}, \qquad \text{mask}_t \in \{0,1\}^{C}
\]
\item A m\'ascara modula a mem\'oria:
\[
\gamma_t = W(1-\text{mask}_t) + b,
\qquad m_t = \gamma_t \odot h_{t-1}
\]
\item Atualiza\c{c}\~ao recorrente:
\[
h_t = \mathrm{KAN}_{cell}\big([\,x_t,\; h_{t-1},\; m_t\,]\big)
\]
\item A sa\'ida \'e a reconstru\c{c}\~ao do pr\'oprio $x_t$:
\[
\hat{x}_t = W_{out} h_t + b_{out}
\]
\item Loss no modo atual: reconstru\c{c}\~ao (MSE) com termo de difus\~ao opcional.
\end{itemize}
\end{frame}

\begin{frame}{Resumo t\'ecnico}
\begin{itemize}
\item O modelo combina \textbf{entrada atual} com \textbf{mem\'oria passada} para reconstruir a s\'erie.
\item A m\'ascara indica faltantes e \textbf{modula} a mem\'oria:
$m_t$ refor\c{c}a/atenua partes de $h_{t-1}$.
\item A c\'elula recorrente \textbf{junta} $x_t$, $h_{t-1}$ e $m_t$ para produzir o novo estado $h_t$.
\item A sa\'ida $\hat{x}_t$ \'e lida do estado via \textbf{head linear} ($W_{out}h_t+b_{out}$).
\item No wrapper \texttt{TSDF\_seqKANSeq}: \texttt{direct\_x=True}, sem \texttt{bi\_gru}, sem cabe\c{c}as de miss/VAE e sem \texttt{log\_likelihood}.
\end{itemize}
\end{frame}

\section{Metodologia}

\begin{frame}{ArticleKAN: splines por conex\~ao}
\small
\begin{itemize}
\item O backbone usa o \textbf{ArticleKAN}: cada aresta $i\!\to\! j$ aprende uma spline univariada.
\item Forma geral:
\[
y_j = \sum_i f_{ij}(x_i), \qquad f_{ij} \text{ spline 1D}
\]
\item Cada spline \'{e} definida por:
\begin{itemize}
\item \textbf{grid} (n\'os) e ordem $k$
\item coeficientes das bases B-spline
\end{itemize}
\item Vantagens:
\begin{itemize}
\item interpretabilidade por conex\~ao (curva $f_{ij}$)
\item controle estrutural da malha de conex\~oes
\item n\~ao-linearidade local por partes
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Vis\~ao geral: SeqKAN recorrente}
\small
Arquitetura por passo de tempo $t$ (com $H=\texttt{hidden\_size}$)
\[
\boxed{
\begin{aligned}
\gamma_t &= W(1-\mathrm{mask}_t) + b \in \mathbb{R}^{H} \\
m_t &= \gamma_t \odot h_{t-1} \in \mathbb{R}^{H} \\
h_t &= \mathrm{KAN}_{cell}\big([x_t,\; h_{t-1},\; m_t]\big) \in \mathbb{R}^{H} \\
\hat{x}_t &= W_{out}h_t + b_{out} \in \mathbb{R}^{C}
\end{aligned}}
\]

\begin{itemize}
\item \textbf{KAN\_cell}: recebe um vetor de dimens\~ao $C+2H$ ($x_t$, $h_{t-1}$ e $m_t$) e atualiza a mem\'oria recorrente.
\item \textbf{Head linear}: reconstru\c{c}\~ao $\hat{x}_t$ a partir do estado.
\end{itemize}
\end{frame}

% --------------------------------------------------
\begin{frame}{M\'ascara $\rightarrow$ mem\'oria modulada}
\small

A m\'ascara gera um gate em hidden via proje\c{c}\~ao linear:
\[
\gamma_t = W(1-\mathrm{mask}_t) + b
\]
e modula o estado anterior:
\[
m_t = \gamma_t \odot h_{t-1}.
\]
Este termo $m_t$ entra como \textbf{terceiro bloco} da entrada da c\'elula:
\[
h_t = \mathrm{KAN}_{cell}([x_t,\; h_{t-1},\; m_t]).
\]

\end{frame}

% =========================
% (4) Top-k — corrigido para o que existe (activation top-k)
% =========================
\begin{frame}{Top-k: esparsidade por ativa\c{c}\~ao ($k_h$)}
\small
\begin{itemize}
\item Existe top-k \textbf{por ativa\c{c}\~ao} aplicado ao vetor $h_t$:
\[
h_t \leftarrow \mathrm{TopK}(h_t; k_h)
\]
mantendo apenas as $k_h$ maiores componentes em magnitude ($|\cdot|$) por amostra/tempo.
\item Warmup: o top-k s\'o entra quando \texttt{current\_epoch > warmup\_epochs}.

\item \textbf{Bottleneck din\^amico} no hidden, reduzindo complexidade e aumentando interpretabilidade por dimens\~ao ativa.
\end{itemize}



\end{frame}

\begin{frame}{Treino no TSDF\_seqKANSeq (c\'odigo atual)}
\small
\begin{itemize}
\item Perda efetiva no modo KAN-only:
\[
\mathcal{L} \approx \lambda_0\,\mathcal{L}_{recon} + \lambda_1\,\mathcal{L}_{noise}
\]
\item $\mathcal{L}_{recon}$: erro de reconstru\c{c}\~ao de $\hat{x}$ (MSE mascarado por observa\c{c}\~oes).
\item $\mathcal{L}_{noise}$: predi\c{c}\~ao de ru\'ido de difus\~ao, usada apenas quando $\lambda_1>0$.
\item Restri\c{c}\~oes impostas no construtor: \texttt{direct\_x=True}, \texttt{static\_dim=0}, \texttt{status\_dim=0}, \texttt{log\_likelihood=False}, e $\lambda_3=\lambda_4=\lambda_5=0$.
\end{itemize}
\end{frame}

\begin{frame}{Interpretabilidade no KAN: leitura por aresta (edge-wise)}
\small
\begin{itemize}
\item Em KAN, cada conex\~ao $i\!\to\! j$ \'{e} uma fun\c{c}\~ao univariada aprendida $f_{ij}(\cdot)$ (spline):
\[
y_j = \sum_i f_{ij}(u_i)
\]
\item No modelo atual existe um conjunto de fun\c{c}\~oes spline inspecion\'aveis:
\begin{itemize}
\item $\mathrm{KAN}_{cell}$: $f^{cell}_{j\to r}([x,h,m]_{t,j})$ (din\^amica no hidden)
\item Sa\'ida linear: contribui\c{c}\~ao por peso $W_{out}(c,j)\cdot h_{t,j}$ (hidden $\to$ feature reconstru\'ida)
\end{itemize}
\item Estrat\'egia de explica\c{c}\~ao recomendada: selecionar arestas mais relevantes (por $|\text{coef}|$ ou contribui\c{c}\~ao) e plotar suas splines.
\end{itemize}
\end{frame}

% =========================
% (6) Conclusão
% =========================
\begin{frame}{Conclus\~ao (alinhada ao c\'odigo atualizado)}
O TSDF\_seqKANSeq (KAN + sa\'ida linear) implementa reconstru\c{c}\~ao recorrente com KAN\_cell e head linear:
\[
m_t=\gamma_t\odot h_{t-1},\quad
\]
\[
h_t=\mathrm{KAN}_{cell}([x_t,h_{t-1},m_t]),\quad
\]
\[
\hat{x}_t=W_{out}h_t+b_{out}
\]

\begin{itemize}
\item N\~ao-linearidade distribu\'ida nas arestas (B-splines do \texttt{ArticleKAN}) na \textbf{c\'elula}; sa\'ida \'{e} \textbf{linear}.
\item M\'ascara atua como \textbf{condicionamento} via $m_t$ (mem\'oria modulada).
\item Esparsidade: \textbf{Top-k por ativa\c{c}\~ao} em $h_t$ (par\^ametro $k_h$); top-k estrutural est\'a desligado.
\item Difus\~ao \'e \textbf{opcional} ($\lambda_1>0$); no modo padr\~ao, a reconstru\c{c}\~ao domina.
\end{itemize}
\end{frame}
