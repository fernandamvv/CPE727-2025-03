\section{Resumo}

\begin{frame}{Resumo (compatível com o código atual)}
\begin{itemize}
\item Modelo: \textbf{TSDF\_seqKANSeq} em modo \textbf{KAN-only}:
      sem MLP decoder, sem VAE, sem miss-head, sem status/static.
\item Entrada efetiva do backbone (por timestep): \textbf{concatenação explícita}
\[
s_t = [x_t, m_t] \in \mathbb{R}^{2C}
\]
onde $m_t$ é máscara feature-wise (1=observado, 0=ausente). No código: \texttt{h = cat([x, mask], -1)}.
\item Célula recorrente: \textbf{SeqKANSeq} computa
\[
h_t = \mathrm{KAN}_{cell}\big([s_t, h_{t-1}]\big), \quad
\hat{x}_t = \mathrm{KAN}_{out}(h_t)
\]
\item Treino supervisionado por reconstrução (MSE). O bloco de difusão injeta ruído em \textbf{[x,mask]} (opcionalmente ruído também na máscara).
\end{itemize}
\end{frame}

\section{Metodologia}



\begin{frame}{KAN: ideia central}
\begin{itemize}
\item Em redes usuais (MLP), cada neurônio combina entradas por pesos lineares:
\[
y_j = \sum_i w_{ij}\, z_i
\]
\item No \textbf{KAN}, cada conexão $i\to j$ é uma \textbf{função univariada aprendida}:
\[
y_j = \sum_i f_{ij}(z_i)
\]
\item Consequências: estrutura \textbf{aditiva}, maior \textbf{interpretabilidade por conexão} e controle direto do grafo.
\end{itemize}

\medskip
\begin{block}{Intuição}
\centering
\textit{Troca “peso” por “função 1D” em cada aresta.}
\end{block}
\end{frame}



% --------------------------------------------------
\begin{frame}{KAN\_cell e KAN\_out: o que são esses dois blocos?}
\begin{itemize}
\item O SeqKANSeq é composto por dois módulos KAN com papéis diferentes:
\begin{itemize}
\item \textbf{KAN\_cell (transição do estado)}: atualiza a memória do modelo.
\item \textbf{KAN\_out (leitura/saída)}: transforma a memória em saída/reconstrução.
\end{itemize}

\item \textbf{KAN\_cell} implementa a dinâmica recorrente:
\[
h_t = \mathrm{KAN}_{cell}\big([s_t,\, h_{t-1}]\big),
\qquad s_t=[x_t,m_t]
\]
Intuição: “como a entrada atual e o estado anterior se combinam para formar o novo estado”.

\item \textbf{KAN\_out} implementa o head de reconstrução:
\[
\hat{x}_t = \mathrm{KAN}_{out}(h_t)
\]
Intuição: “como ler o estado oculto para reconstruir as features”.

\item Ambos são \textbf{KAN}: em vez de pesos lineares, usam somas de \textbf{funções univariadas por conexão},
o que facilita controle estrutural (ex.: Top-k) e interpretabilidade.
\item Diferença para GRU/LSTM: não há gates multiplicativos; a não linearidade está nas \textbf{funções das conexões}.
\end{itemize}
\end{frame}



% --------------------------------------------------
\begin{frame}{O que são as splines no KAN }
\begin{itemize}
\item No KAN, cada conexão $i\to j$ aprende uma função univariada $f_{ij}(\cdot)$.
Em vez de escolher uma ativação fixa (ReLU, tanh), o modelo \textbf{aprende o formato da curva}.
\item Essa função é parametrizada como uma \textbf{spline}:
uma curva \textbf{por partes} definida em intervalos do eixo-$x$.
\item O domínio é dividido em \textbf{nós} (grid). Em cada intervalo entre nós, a spline usa um polinômio simples,
mas a junção entre intervalos é feita de forma \textbf{suave} (continuidade).
\item Intuição prática:
\begin{itemize}
\item mais nós $\Rightarrow$ mais flexibilidade para capturar não linearidades;
\item menos nós $\Rightarrow$ função mais simples e mais estável.
\end{itemize}
\item Interpretabilidade: ao inspecionar a curva $f_{ij}(u)$, vemos \textbf{como uma entrada específica influencia} o neurônio de saída.
\end{itemize}
\end{frame}


% --------------------------------------------------
% --------------------------------------------------
\begin{frame}{Splines no KAN: função aprendida por conexão (k=3)}
\begin{itemize}
\item No KAN, cada conexão $i\to j$ aprende uma função univariada
\[
f_{ij}:\mathbb{R}\rightarrow\mathbb{R},
\qquad u \mapsto f_{ij}(u)
\]
\item Em vez de uma ativação fixa (ReLU/tanh), $f_{ij}$ é parametrizada como uma \textbf{B-spline}:
\[
f_{ij}(u)=\sum_{m=1}^{M}\alpha_{ijm}\,B_m^{(3)}(u)
\]
\item Interpretação dos termos:
\begin{itemize}
\item $B_m^{(3)}(u)$: bases B-spline de \textbf{ordem 3} (grau 2, quadráticas),
\item $\alpha_{ijm}$: coeficientes aprendidos por treino,
\item $M$: número de bases (depende do número de nós no grid).
\end{itemize}
\item Intuição: cada aresta aprende uma \textbf{curva suave} $u\mapsto f_{ij}(u)$, inspecionável e controlável.
\end{itemize}
\end{frame}
% --------------------------------------------------
% --------------------------------------------------
\begin{frame}{Nós (grid), suporte local e suavidade (k=3)}
\begin{itemize}
\item A spline é definida a partir de uma sequência de nós (grid)
\[
t_0 \le t_1 \le \cdots \le t_{K},
\qquad u \in [t_0,t_{K}]
\]
\item \textbf{Suporte local}: cada base afeta apenas uma região do eixo-$u$:
\[
B_m^{(3)}(u)=0 \quad \text{fora de } [t_m,\; t_{m+3}]
\]
\item \textbf{Suavidade}: com $k=3$ (grau 2), a função aprendida é tipicamente
\[
f_{ij} \in C^{1}
\]
isto é, $f_{ij}$ e sua primeira derivada são contínuas (exceto se houver nós repetidos).
\item Trade-off do grid:
\begin{itemize}
\item mais nós $\Rightarrow$ maior flexibilidade para capturar não linearidades finas;
\item menos nós $\Rightarrow$ função mais simples e mais estável.
\end{itemize}
\end{itemize}

\medskip
\begin{block}{Conexão com o SeqKANSeq}
\centering
Essas splines parametrizam as conexões tanto na \(\mathrm{KAN}_{cell}\) (dinâmica) quanto na \(\mathrm{KAN}_{out}\) (reconstrução).
\end{block}
\end{frame}

% --------------------------------------------------
\begin{frame}{Esparsidade Estrutural (Top-k) por neurônio}
\begin{itemize}
\item Para aumentar \textbf{interpretabilidade} e \textbf{controle estrutural}, aplicamos \textbf{Top-k} nas conexões do KAN.
\item Como o KAN é aditivo por arestas, cada neurônio $j$ recebe contribuições por conexões $i\to j$:
\[
y_j = \sum_i f_{ij}(z_i)
\]
\item Com Top-k \textbf{por neurônio}, mantemos apenas as $k$ conexões mais relevantes \textbf{para cada neurônio $j$}:
\[
y_j = \sum_{i \in \mathcal{S}_j} f_{ij}(z_i),
\qquad |\mathcal{S}_j| \le k
\]
\item Interpretação: cada neurônio passa a depender de um \textbf{conjunto pequeno e estável de entradas},
o que simplifica a leitura do grafo e reduz conexões fracas/ruidosas.
\end{itemize}

\medskip
\begin{block}{Onde aplicamos}
\centering
Na célula recorrente ($[x_t,m_t]$ e $h_{t-1}$) e/ou no head de saída ($h_t \rightarrow \hat{x}_t$).
\end{block}
\end{frame}





% --------------------------------------------------
\begin{frame}{Top-k na célula recorrente (entrada vs memória)}
\begin{itemize}
\item Na atualização do estado, o SeqKANSeq combina \textbf{entrada} e \textbf{memória}:
\[
h_t = \mathrm{KAN}_{cell}\big([x_t, m_t, h_{t-1}]\big)
\]
\item Com \textbf{Top-k por neurônio}, cada neurônio oculto seleciona apenas um número limitado de conexões relevantes.
\item Essa seleção pode ser feita \textbf{separando as fontes}:
\begin{itemize}
\item até $k_x$ conexões vindas da \textbf{entrada} ($[x_t,m_t]$),
\item até $k_h$ conexões vindas da \textbf{memória} ($h_{t-1}$),
\item (opcional) tratar a \textbf{máscara} $m_t$ separadamente para não competir com as features de $x_t$.
\end{itemize}
\item Intuição: controlar \textbf{quantas variáveis do instante} e \textbf{quanta memória passada} cada neurônio pode usar.
\end{itemize}
\end{frame}

% --------------------------------------------------
% --------------------------------------------------
\begin{frame}{Top-k no head de saída (reconstrução)}
\begin{itemize}
\item A reconstrução é produzida por um head (KAN) que lê o estado:
\[
\hat{x}_t = \mathrm{KAN}_{out}(h_t)
\]
\item Com \textbf{Top-k por saída}, cada componente $\hat{x}_{t,k}$ depende de apenas algumas dimensões de $h_t$:
isso limita quantos estados ocultos influenciam cada feature reconstruída.
\item Em todos os casos, o Top-k atua como \textbf{máscara estrutural}: ele \textbf{remove conexões} sem mudar a forma das funções nas conexões mantidas.
\item A seleção respeita as conexões permitidas pelo modelo-base (máscara estrutural inicial).
\end{itemize}
\end{frame}

\begin{frame}{Conclusão (alinhada ao código)}
O TSDF\_seqKANSeq (KAN-only) implementa:
\[
h_t = F([x_t,m_t], h_{t-1}), \quad \hat{x}_t = G(h_t)
\]

com:
\begin{itemize}
\item Não linearidade distribuída nas arestas (B-splines do ArticleKAN)
\item Recorrência explícita sem gates multiplicativos (estilo GRU/LSTM)
\item Top-k estrutural opcional aplicado à \textbf{máscara de conexões} após warmup
\item Separação explícita do top-k na célula: $x$ vs $m$ vs $h_{t-1}$ (quando configurado)
\end{itemize}

Resultados mostram gap em relação ao GRU nas configs atuais,
mas com maior controle estrutural e interpretabilidade por aresta.
\end{frame}
